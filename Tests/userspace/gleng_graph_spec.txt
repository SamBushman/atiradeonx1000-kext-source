# ---- PPStream (the parser/optimiser stream the _glePPPrune* functions walk) --------------------------------------------------------------------------------------------------
# ctx+0x1c -> stream; stream+4 locals, +0x10 params, +0x14 param arrays, +0x18 attribs, +0x1c outputs, +0x20 operations: each a pointer to a list head (+0 first node, +8 count);
# a chunk node: +8 next, +0xc.. binding words, +0xe lo index, +0x12 index, +0x14 flag; an operation node: +0xc header (>>0x1d operand count, >>0x12&0xff opcode), operand words at +0x14,+0x1c,+0x24,+0x2c
# (type in bits 0x11/0x17..0x19, sign bit, index shorts at +0x1a,+0x22,+0x2a,+0x32). The words come from small pick sets (count 0..3 / opcode 0, 0x10, 0x44, 0x4a; operand type 0..3 with and without the sign bit)
# and the indexes are drawn from 0..3, so the equal-index and type tests of all five functions are taken often (a fully random word takes one of them in ~2% of the nodes).

  pick 0x0c 0x20000000 0x20400000 0x21280000 0x21100000 0x40000000 0x40400000 0x41100000 0x60000000

  u16 0x12 0 3
  pick 0x14 0 1 0x00020000 0x00060000 0x00040000
  pick 0x1c 0 0x00800000 0x01000000 0x01800000 0x80000000 0x81000000
  pick 0x24 0 0x00800000 0x01000000 0x01800000 0x80000000 0x81000000
  pick 0x2c 0 0x00800000 0x01000000 0x01800000 0x80000000 0x81000000
  u16 0x1a 0 3

# ---- shader / program objects (ARB_shader_objects): the machine-object hash table, a program object, a shader object --------------------------------------------------------------

# +0x1000 word (SS+0x7018) is the free-name range list), +0x7594/+0x7598 (the compile/link option words handed to the Sh* library). A hashed object: +0 next, +4 name, +8 refcount, +0xc free callback,

# ---- _glePPParse: the assembly-program parser front end ----------------------------------------------------------------------------------------------------------------------------
# The PP* imports hand back library objects (parser, stream, chunk lists); here they are declared as graphs (`ret`) or modelled in fnfuzz.c gs_semantic (PPParserAttachStream sets parser+0x1c,
# PPStreamChunkListAddChunk appends, ...CreateFromChunkList / ...CreateFromChunk clone, ...GetStream reports the node count). The parser is the object the _glePPPrune* functions read
# (+0x1c stream, +0x442 flag byte, +0x464 / +0x468 limits); the program state `param_2` holds the stream (+0), the source (+4), a result array (+0x4bc) and the PP emulator program (+0x4d4).
type PPLIST0 0x10
# the parse succeeds only when every _glePPPrune* passes its count limit (stream list count <= parser +0x464 / +0x468): lists of 0..3 nodes, limits 4..8
type PPLIST_P 0x10

  u32 0x08 0 3

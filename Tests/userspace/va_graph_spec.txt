# va_graph_spec.txt - object-graph specs for the VADriver render-pass functions (FNFUZZ_GRAPHSPEC, Tests/userspace/fnfuzz.c; syntax in the comment above load_graphspec).
# The `ava_render_*_pass` functions take the renderer context (command-buffer write pointer ctx+0x34, limit ctx+0x38, last header ctx+0x30, surface sizes ctx+0x578 / 0x57c /
# 0x588, the video-adjustment floats ctx+0x12c hue, 0x130 contrast, 0x134 brightness, 0x138 saturation) and finish with _IOConnectMapMemory (a stubbed import under FNFUZZ_INTERPOSE=1).
# The command buffer words are part of the arena digest, and ava_emit_shader_constants copies the hueSaturation / brightnessContrast constant words into it, so a wrong constant is visible.
type VAHDR 0x20
type VACMD 0x800
type VACTX 0x2000
  ptr 0x30 VAHDR
  ptr 0x34 VACMD
  padd 0x38 0x34 0x400 0x600
  u32 0x1010 0 0x40                  # the mapping the stubbed _IOConnectMapMemory `returns` is ctx+0x1000 (see `out` below): its word +0x10 is the entry count
  u32 0x14 0 100
  u32 0xac4 0 3
  pick 0x578 0x40 0x100 0x2d0 0x400
  pick 0x57c 0x40 0x100 0x1e0 0x240
  pick 0x588 0x80 0x200 0x2d0 0x400
  pick 0x12c 0 0x3f800000 0xbf800000 0x40490fdb 0xc0490fdb 0x3dcccccd 0x3f000000 0x3e800000 0xbe99999a 0x40000000
  pick 0x130 0x3f800000 0 0x40400000 0x3f000000 0x3e000000 0x40000000 0x3f99999a
  pick 0x134 0 0x3e4ccccd 0xbe4ccccd 0x3f000000 0xbf800000 0x3f800000
  pick 0x138 0x3f800000 0x3e800000 0x40800000 0x3f000000 0x40000000 0x3fc00000
func ava_render_hue_saturation_pass
  arg 0 ptr VACTX
  arg 1 rand
  arg 2 pick 0 1
  out IOConnectMapMemory 3 0x50002000     # *(ctx+0x40) = ctx+0x1000 (the first graph node is the context, at ARENA_BASE+0x1000)
func ava_render_brightness_contrast_pass
  arg 0 ptr VACTX
  arg 1 rand
  arg 2 pick 0 1
  out IOConnectMapMemory 3 0x50002000
